{%- comment -%} Navigation, generated from front matter. Nothing here is hand-maintained: a new document appears once it has a `section` (injected by _config.yml defaults from its directory) and an `order`. Deprecated and superseded documents drop out of nav by design. They stay reachable by direct link so citations do not rot. {%- endcomment -%}
{%- for s in site.sections -%} {%- assign docs = site.pages | where: "section", s.key | where_exp: "d", "d.status != 'deprecated' and d.status != 'superseded'" | sort: "order" -%} {%- if docs.size > 0 -%}

{{ s.title }}

{%- assign groups = docs | map: "group" | compact | uniq -%} {%- if groups.size == 0 -%}
    {%- for doc in docs -%} {{ doc.title }} {%- endfor -%}
{%- else -%} {%- for g in groups -%}

{{ g }}

    {%- assign in_group = docs | where: "group", g -%} {%- for doc in in_group -%} {{ doc.title }} {%- endfor -%}
{%- endfor -%} {%- endif -%}
{%- endif -%} {%- endfor -%}
